            const y = d.getFullYear();

        function parseDate(str) {
            if (!str) return null;
            const parts = str.split('-');
            if (parts.length === 3) return new Date(parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2]));
            return null;

        function saveFocusData() { localStorage.setItem('focusData_v1', JSON.stringify(focusData)); }

        function addFocusMinutes(minutes) {
            const fd = getTodayFocusData();
            fd.totalMinutes += minutes;
            const newRewardCount = Math.floor(fd.totalMinutes / 60);
            let earnedReward = 0;
            if (newRewardCount > fd.rewardCount) {
                earnedReward = (newRewardCount - fd.rewardCount) * 5;
                fd.rewardCount = newRewardCount;
            }
            saveFocusData();
            return { earnedReward, totalMinutes: fd.totalMinutes, rewardCount: fd.rewardCount };
        }

        function getFocusProgress() {
            const fd = getTodayFocusData();
            const remainInCycle = fd.totalMinutes % 60;
            const progressPercent = (remainInCycle / 60) * 100;
            const remainToNext = 60 - remainInCycle;
            return { totalMinutes: fd.totalMinutes, rewardCount: fd.rewardCount, remainToNext: remainToNext,
                progressPercent: progressPercent, totalRewardEarned: fd.rewardCount * 5 };
        }

        function analyzeTaskDifficulty(text) {
            if (!text || !text.trim()) return { level: 'mid', reward: 2, label: '普通' };
            const lower = text.trim().toLowerCase();
            if (/重要|紧急|难|复杂|挑战|大量|深入|全面|关键|核心|重大|严格|繁重|攻坚|突破|重构|架构/.test(lower)) return { level: 'high',
                reward: 10, label: '困难' };
            if (/学习|完成|写|做|开发|设计|处理|准备|研究|规划|安排|制作|编写|测试|部署|整理|优化|改进|升级|修复|实现|构建/.test(lower)) return { level: 'mid',
                reward: 5, label: '中等' };
            if (/简单|快速|轻松|顺便|小|微|浅|略|随便|稍微|简单看看|瞄一眼/.test(lower)) return { level: 'low', reward: 1, label: '简单' };
            if (text.trim().length < 8) return { level: 'low', reward: 1, label: '简单' };
            if (text.trim().length < 20) return { level: 'mid', reward: 2, label: '普通' };
            if (text.trim().length < 35) return { level: 'mid', reward: 5, label: '中等' };
            return { level: 'high', reward: 10, label: '困难' };
        }
